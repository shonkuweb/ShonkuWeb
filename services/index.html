Agent Execution Architecture
// Customer Support Agent Workflow
const agent = new ShonkuAgent({
  role: "Tier-1 Autonomous Specialist",
  vectorStore: EnterpriseKnowledgeBase,
  guardrails: ["strict_truthfulness", "pii_masking"],
  tools: [
    tools.orderLookup,
    tools.modifyShipping,
    tools.humanEscalation
  ]
});

// Event Triggered
agent.onMessage(async (inquiry) => {
  const ctx = await agent.retrieve(inquiry);
  const decision = await agent.reason(ctx);
  return agent.executeAndReply(decision);
});
